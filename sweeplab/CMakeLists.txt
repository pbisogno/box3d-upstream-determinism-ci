cmake_minimum_required(VERSION 3.22)
project(box3d_public_sweep_experiment LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static library" FORCE)
set(BOX3D_PROFILE OFF CACHE BOOL "No profiler dependency" FORCE)
set(BOX3D_VALIDATE ON CACHE BOOL "Upstream validation" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(SWEEPLAB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(SWEEPLAB_UPSTREAM "${SWEEPLAB_ROOT}/upstream")
add_subdirectory("${SWEEPLAB_UPSTREAM}" upstream-build)

function(sweeplab_contraction target)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        if(CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
            target_compile_options("${target}" PRIVATE /clang:-ffp-contract=off)
        else()
            target_compile_options("${target}" PRIVATE -ffp-contract=off)
        endif()
    endif()
endfunction()

# Each library owns generated source copies. The original target and upstream
# checkout are untouched. Experimental zero-policy clones are explicitly named.
function(sweeplab_variant name generator)
    set(generated "${CMAKE_BINARY_DIR}/${name}-generated")
    execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/${generator}"
        "${SWEEPLAB_UPSTREAM}" "${generated}" ${ARGN}
        RESULT_VARIABLE result OUTPUT_VARIABLE manifest)
    if(NOT result EQUAL 0)
        message(FATAL_ERROR "Pinned generation failed: ${generator}")
    endif()
    message(STATUS "${manifest}")
    get_target_property(source_dir box3d SOURCE_DIR)
    get_target_property(original_sources box3d SOURCES)
    set(sources "")
    foreach(source IN LISTS original_sources)
        if(NOT IS_ABSOLUTE "${source}")
            set(source "${source_dir}/${source}")
        endif()
        get_filename_component(base "${source}" NAME)
        if(EXISTS "${generated}/${base}")
            set(source "${generated}/${base}")
        endif()
        list(APPEND sources "${source}")
    endforeach()
    add_library("${name}" STATIC ${sources})
    foreach(property IN ITEMS INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS LINK_LIBRARIES
        INTERFACE_INCLUDE_DIRECTORIES INTERFACE_COMPILE_DEFINITIONS INTERFACE_LINK_LIBRARIES
        C_STANDARD C_STANDARD_REQUIRED C_EXTENSIONS MSVC_RUNTIME_LIBRARY)
        get_target_property(value box3d "${property}")
        if(NOT value STREQUAL "value-NOTFOUND")
            set_property(TARGET "${name}" PROPERTY "${property}" "${value}")
        endif()
    endforeach()
    target_include_directories("${name}" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${source_dir}")
    sweeplab_contraction("${name}")
    if(MSVC AND BOX3D_AVX2 AND NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
        if(EXISTS "${generated}/wide8.c")
            set_source_files_properties("${generated}/wide8.c" PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
        else()
            set_source_files_properties("${source_dir}/wide8.c" PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
        endif()
    endif()
endfunction()

sweeplab_variant(sweeplab-ccd-observed instrument_ccd.py)
sweeplab_variant(sweeplab-zero-identity-library instrument_zero.py --mode 0)
sweeplab_variant(sweeplab-zero-canonical-library instrument_zero.py --mode 1)

function(sweeplab_executable name source library observed)
    add_executable("${name}" "${SWEEPLAB_ROOT}/${source}")
    set_target_properties("${name}" PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO
        MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_include_directories("${name}" PRIVATE "${SWEEPLAB_UPSTREAM}/src" "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries("${name}" PRIVATE "${library}" Threads::Threads)
    target_compile_definitions("${name}" PRIVATE SWEEPLAB_INSTRUMENTED=${observed} CONTACTLAB_INSTRUMENTED=${observed})
    if(NOT BOX3D_AVX2)
        target_compile_definitions("${name}" PRIVATE BOX3D_DISABLE_AVX2)
    endif()
    if(BOX3D_DISABLE_SIMD)
        target_compile_definitions("${name}" PRIVATE BOX3D_DISABLE_SIMD)
    endif()
    sweeplab_contraction("${name}")
endfunction()

sweeplab_executable(sweeplab-zero-reference contactlab-zero.c box3d::box3d 0)

# The microtest calls the actual W8 helper; its translation unit needs AVX2
# instructions only in the matching runtime-checked x64 profile.
if(BOX3D_AVX2)
    if(MSVC AND NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
        target_compile_options(sweeplab-operator PRIVATE /arch:AVX2)
    elseif(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        if(CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
            target_compile_options(sweeplab-operator PRIVATE /clang:-mavx2)
        else()
            target_compile_options(sweeplab-operator PRIVATE -mavx2)
        endif()
    endif()
endif()
