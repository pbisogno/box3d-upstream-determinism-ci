cmake_minimum_required(VERSION 3.22)
project(box3d_public_contact_lab LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static library" FORCE)
set(BOX3D_PROFILE OFF CACHE BOOL "No profiler dependency" FORCE)
set(BOX3D_VALIDATE ON CACHE BOOL "Upstream validation" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(CONTACTLAB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(CONTACTLAB_UPSTREAM "${CONTACTLAB_ROOT}/upstream")
add_subdirectory("${CONTACTLAB_UPSTREAM}" upstream-build)

# Every observation variant is a separate clone. The original target and its
# input files remain unchanged, and each driver has a pristine executable twin.
function(contactlab_variant name generator)
    set(generated "${CMAKE_BINARY_DIR}/${name}-generated")
    execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/${generator}"
        "${CONTACTLAB_UPSTREAM}" "${generated}"
        RESULT_VARIABLE result OUTPUT_VARIABLE manifest)
    if(NOT result EQUAL 0)
        message(FATAL_ERROR "Pinned observation generation failed: ${generator}")
    endif()
    message(STATUS "${manifest}")
    get_target_property(source_dir box3d SOURCE_DIR)
    get_target_property(original_sources box3d SOURCES)
    set(sources "")
    foreach(source IN LISTS original_sources)
        if(NOT IS_ABSOLUTE "${source}")
            set(source "${source_dir}/${source}")
        endif()
        get_filename_component(base "${source}" NAME)
        if(EXISTS "${generated}/${base}")
            set(source "${generated}/${base}")
        endif()
        list(APPEND sources "${source}")
    endforeach()
    add_library("${name}" STATIC ${sources})
    foreach(property IN ITEMS INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS LINK_LIBRARIES
        INTERFACE_INCLUDE_DIRECTORIES INTERFACE_COMPILE_DEFINITIONS INTERFACE_LINK_LIBRARIES
        C_STANDARD C_STANDARD_REQUIRED C_EXTENSIONS MSVC_RUNTIME_LIBRARY)
        get_target_property(value box3d "${property}")
        if(NOT value STREQUAL "value-NOTFOUND")
            set_property(TARGET "${name}" PROPERTY "${property}" "${value}")
        endif()
    endforeach()
    target_include_directories("${name}" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${source_dir}")
    # The upstream directory's contraction option does not propagate upward.
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        if(CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
            target_compile_options("${name}" PRIVATE /clang:-ffp-contract=off)
        else()
            target_compile_options("${name}" PRIVATE -ffp-contract=off)
        endif()
    endif()
    if(MSVC AND BOX3D_AVX2 AND NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
        if(EXISTS "${generated}/wide8.c")
            set_source_files_properties("${generated}/wide8.c" PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
        else()
            set_source_files_properties("${source_dir}/wide8.c" PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
        endif()
    endif()
endfunction()

contactlab_variant(contactlab-physics-hooks instrument_physics.py)
contactlab_variant(contactlab-zero-hooks instrument_zero.py)

foreach(driver IN ITEMS physics physics-pristine zero zero-pristine)
    string(REPLACE "-pristine" "" source_driver "${driver}")
    add_executable("contactlab-${driver}" "${CONTACTLAB_ROOT}/contactlab-${source_driver}.c")
    set_target_properties("contactlab-${driver}" PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO
        MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_include_directories("contactlab-${driver}" PRIVATE "${CONTACTLAB_UPSTREAM}/src" "${CMAKE_CURRENT_SOURCE_DIR}")
    if(driver MATCHES "-pristine$")
        target_link_libraries("contactlab-${driver}" PRIVATE box3d::box3d Threads::Threads)
        target_compile_definitions("contactlab-${driver}" PRIVATE CONTACTLAB_INSTRUMENTED=0)
    else()
        target_link_libraries("contactlab-${driver}" PRIVATE "contactlab-${driver}-hooks" Threads::Threads)
        target_compile_definitions("contactlab-${driver}" PRIVATE CONTACTLAB_INSTRUMENTED=1)
    endif()
    if(NOT BOX3D_AVX2)
        target_compile_definitions("contactlab-${driver}" PRIVATE BOX3D_DISABLE_AVX2)
    endif()
    if(BOX3D_DISABLE_SIMD)
        target_compile_definitions("contactlab-${driver}" PRIVATE BOX3D_DISABLE_SIMD)
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        if(CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
            target_compile_options("contactlab-${driver}" PRIVATE /clang:-ffp-contract=off)
        else()
            target_compile_options("contactlab-${driver}" PRIVATE -ffp-contract=off)
        endif()
    endif()
endforeach()
