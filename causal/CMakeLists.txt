cmake_minimum_required(VERSION 3.22)
project(box3d_causal_observation LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static library" FORCE)
set(BOX3D_PROFILE OFF CACHE BOOL "No profiler dependency" FORCE)
set(BOX3D_VALIDATE ON CACHE BOOL "Upstream validation" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../upstream" upstream-build)
add_executable(causal-replay "${CMAKE_CURRENT_SOURCE_DIR}/../causal-replay.c")
set_target_properties(causal-replay PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
# The opt-in diagnostic TU accesses published internal cache definitions.
# This does not change any upstream source or exported library ABI.
target_include_directories(causal-replay PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../upstream/src")
target_link_libraries(causal-replay PRIVATE box3d::box3d)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    if(CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
        target_compile_options(causal-replay PRIVATE /clang:-ffp-contract=off)
    else()
        target_compile_options(causal-replay PRIVATE -ffp-contract=off)
    endif()
endif()
