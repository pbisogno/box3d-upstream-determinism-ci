cmake_minimum_required(VERSION 3.22)
project(box3d_public_replay LANGUAGES C CXX)
# Build the unchanged upstream checkout as a dependency, without optional packages.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static library" FORCE)
set(BOX3D_PROFILE OFF CACHE BOOL "No profiler dependency" FORCE)
set(BOX3D_VALIDATE ON CACHE BOOL "Upstream debug validation" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_subdirectory(upstream upstream-build)
add_executable(portable-replay portable-replay.c)
set_target_properties(portable-replay PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_link_libraries(portable-replay PRIVATE box3d::box3d)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    if(CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
        target_compile_options(portable-replay PRIVATE /clang:-ffp-contract=off)
    else()
        target_compile_options(portable-replay PRIVATE -ffp-contract=off)
    endif()
endif()
