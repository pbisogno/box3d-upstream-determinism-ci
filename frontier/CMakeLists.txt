cmake_minimum_required(VERSION 3.22)
project(box3d_public_frontiers LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static library" FORCE)
set(BOX3D_PROFILE OFF CACHE BOOL "No profiler dependency" FORCE)
set(BOX3D_VALIDATE ON CACHE BOOL "Upstream validation" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(FRONTIER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(FRONTIER_UPSTREAM "${FRONTIER_ROOT}/upstream")
add_subdirectory("${FRONTIER_UPSTREAM}" upstream-build)

# Keep the pristine library for every ordinary measurement and the cache twin.
# Clone the exact target with only three generated translation units replaced.
set(FRONTIER_GENERATED "${CMAKE_BINARY_DIR}/frontier-generated")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/instrument_cache.py"
    "${FRONTIER_UPSTREAM}" "${FRONTIER_GENERATED}"
    RESULT_VARIABLE frontier_generation_status OUTPUT_VARIABLE frontier_generation_manifest)
if(NOT frontier_generation_status EQUAL 0)
    message(FATAL_ERROR "Pinned cache instrumentation could not be generated")
endif()
message(STATUS "${frontier_generation_manifest}")
get_target_property(frontier_source_dir box3d SOURCE_DIR)
get_target_property(frontier_sources box3d SOURCES)
set(frontier_instrumented_sources "")
foreach(source IN LISTS frontier_sources)
    if(NOT IS_ABSOLUTE "${source}")
        set(source "${frontier_source_dir}/${source}")
    endif()
    get_filename_component(source_name "${source}" NAME)
    if(source_name STREQUAL "contact_solver.c" OR source_name STREQUAL "wide4.c" OR source_name STREQUAL "wide8.c")
        list(APPEND frontier_instrumented_sources "${FRONTIER_GENERATED}/${source_name}")
    elseif(NOT source_name STREQUAL "contact_solver_wide.inl")
        list(APPEND frontier_instrumented_sources "${source}")
    endif()
endforeach()
add_library(box3d-frontier-hooks STATIC ${frontier_instrumented_sources})
foreach(property IN ITEMS INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS LINK_LIBRARIES
    INTERFACE_INCLUDE_DIRECTORIES INTERFACE_COMPILE_DEFINITIONS INTERFACE_LINK_LIBRARIES
    C_STANDARD C_STANDARD_REQUIRED C_EXTENSIONS MSVC_RUNTIME_LIBRARY)
    get_target_property(value box3d "${property}")
    if(NOT value STREQUAL "value-NOTFOUND")
        set_property(TARGET box3d-frontier-hooks PROPERTY "${property}" "${value}")
    endif()
endforeach()
target_include_directories(box3d-frontier-hooks PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${frontier_source_dir}")
# Directory-wide options from the nested upstream directory do not propagate
# to its parent. Apply the same deterministic contraction rule to the clone.
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    if(CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
        target_compile_options(box3d-frontier-hooks PRIVATE /clang:-ffp-contract=off)
    else()
        target_compile_options(box3d-frontier-hooks PRIVATE -ffp-contract=off)
    endif()
endif()
if(MSVC AND BOX3D_AVX2 AND NOT CMAKE_C_COMPILER_ID STREQUAL "Clang")
    set_source_files_properties("${FRONTIER_GENERATED}/wide8.c" PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
endif()

foreach(driver IN ITEMS cache cache-plain rollback concurrency numeric quality)
    if(driver STREQUAL "cache-plain")
        set(source_driver cache)
    else()
        set(source_driver "${driver}")
    endif()
    add_executable("frontier-${driver}" "${FRONTIER_ROOT}/frontier-${source_driver}.c")
    set_target_properties("frontier-${driver}" PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO
        MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_include_directories("frontier-${driver}" PRIVATE "${FRONTIER_UPSTREAM}/src" "${CMAKE_CURRENT_SOURCE_DIR}")
    if(driver STREQUAL "cache")
        target_link_libraries("frontier-${driver}" PRIVATE box3d-frontier-hooks Threads::Threads)
        target_compile_definitions("frontier-${driver}" PRIVATE FRONTIER_INSTRUMENTED=1)
    else()
        target_link_libraries("frontier-${driver}" PRIVATE box3d::box3d Threads::Threads)
    endif()
    if(NOT BOX3D_AVX2)
        target_compile_definitions("frontier-${driver}" PRIVATE BOX3D_DISABLE_AVX2)
    endif()
    if(BOX3D_DISABLE_SIMD)
        target_compile_definitions("frontier-${driver}" PRIVATE BOX3D_DISABLE_SIMD)
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        if(CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
            target_compile_options("frontier-${driver}" PRIVATE /clang:-ffp-contract=off)
        else()
            target_compile_options("frontier-${driver}" PRIVATE -ffp-contract=off)
        endif()
    endif()
endforeach()
if(BOX3D_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|x64)$")
    if(MSVC)
        target_compile_options(frontier-numeric PRIVATE /arch:AVX2)
    else()
        target_compile_options(frontier-numeric PRIVATE -mavx2)
    endif()
endif()
